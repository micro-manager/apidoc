65 class MMTime
66 {
67 long long microseconds_;
68
69 public:
70 MMTime() : microseconds_(0LL) {}
71
72 explicit MMTime(double uSecTotal) :
73 microseconds_(static_cast<long long>(uSecTotal))
74 {}
75
76 explicit MMTime(long sec, long uSec) :
77 microseconds_(sec * 1'000'000LL + uSec)
78 {}
79
80 static MMTime fromUs(long long us)
81 {
82 // Work around our lack of a constructor that directly sets the
83 // internal representation.
84 // (Note that we cannot add a constructor from 'long long' because
85 // many existing uses would then get an error (ambiguous with the
86 // 'double' overload).)
87 MMTime ret;
88 ret.microseconds_ = us;
89 return ret;
90 }
91
92 static MMTime fromMs(double ms)
93 {
94 return MMTime(ms * 1000.0);
95 }
96
97 static MMTime fromSeconds(long secs)
98 {
99 return MMTime(secs, 0);
100 }
101
102 MMTime operator+(const MMTime &other) const
103 {
104 return fromUs(microseconds_ + other.microseconds_);
105 }
106
107 MMTime operator-(const MMTime &other) const
108 {
109 return fromUs(microseconds_ - other.microseconds_);
110 }
111
112 bool operator>(const MMTime &other) const
113 {
114 return microseconds_ > other.microseconds_;
115 }
116
117 bool operator>=(const MMTime &other) const
118 {
119 return microseconds_ >= other.microseconds_;
120 }
121
122 bool operator<(const MMTime &other) const
123 {
124 return microseconds_ < other.microseconds_;
125 }
126
127 bool operator<=(const MMTime &other) const
128 {
129 return microseconds_ <= other.microseconds_;
130 }
131
132 bool operator==(const MMTime &other) const
133 {
134 return microseconds_ == other.microseconds_;
135 }
136
137 bool operator!=(const MMTime &other) const
138 {
139 return !(*this == other);
140 }
141
142 double getMsec() const
143 {
144 return microseconds_ / 1000.0;
145 }
146
147 double getUsec() const
148 {
149 return static_cast<double>(microseconds_);
150 }
151
152 std::string toString() const {
153 long long absUs = std::abs(microseconds_);
154 long long seconds = absUs / 1'000'000LL;
155 long long fracUs = absUs - seconds * 1'000'000LL;
156 const char *sign = microseconds_ < 0 ? "-" : "";
157
158 using namespace std;
159 ostringstream s;
160 s << sign << seconds << '.' <<
161 setfill('0') << right << setw(6) << fracUs;
162 return s.str();
163 }
164 };

170 class TimeoutMs
171 {
172 public:
173 // arguments: MMTime start time, millisecond interval time
174 explicit TimeoutMs(const MMTime startTime, const unsigned long intervalMs) :
175 startTime_(startTime),
176 interval_(0, 1000*intervalMs)
177 {
178 }
179 explicit TimeoutMs(const MMTime startTime, const MMTime interval) :
180 startTime_(startTime),
181 interval_(interval)
182 {
183 }
184
185 bool expired(const MMTime tnow)
186 {
187 MMTime elapsed = tnow - startTime_;
188 return ( interval_ < elapsed );
189 }
190
191 private:
192 MMTime startTime_; // start time
193 MMTime interval_; // interval in milliseconds
194 };
