76 class MMTime
77 {
78 long long microseconds_;
79
80 public:
81 MMTime() : microseconds_(0LL) {}
82
83 explicit MMTime(double uSecTotal) :
84 microseconds_(static_cast<long long>(uSecTotal))
85 {}
86
87 explicit MMTime(long sec, long uSec) :
88 microseconds_(sec * 1'000'000LL + uSec)
89 {}
90
91 static MMTime fromUs(long long us)
92 {
93 // Work around our lack of a constructor that directly sets the
94 // internal representation.
95 // (Note that we cannot add a constructor from 'long long' because
96 // many existing uses would then get an error (ambiguous with the
97 // 'double' overload).)
98 MMTime ret;
99 ret.microseconds_ = us;
100 return ret;
101 }
102
103 static MMTime fromMs(double ms)
104 {
105 return MMTime(ms * 1000.0);
106 }
107
108 static MMTime fromSeconds(long secs)
109 {
110 return MMTime(secs, 0);
111 }
112
113 MMTime operator+(const MMTime &other) const
114 {
115 return fromUs(microseconds_ + other.microseconds_);
116 }
117
118 MMTime operator-(const MMTime &other) const
119 {
120 return fromUs(microseconds_ - other.microseconds_);
121 }
122
123 bool operator>(const MMTime &other) const
124 {
125 return microseconds_ > other.microseconds_;
126 }
127
128 bool operator>=(const MMTime &other) const
129 {
130 return microseconds_ >= other.microseconds_;
131 }
132
133 bool operator<(const MMTime &other) const
134 {
135 return microseconds_ < other.microseconds_;
136 }
137
138 bool operator<=(const MMTime &other) const
139 {
140 return microseconds_ <= other.microseconds_;
141 }
142
143 bool operator==(const MMTime &other) const
144 {
145 return microseconds_ == other.microseconds_;
146 }
147
148 bool operator!=(const MMTime &other) const
149 {
150 return !(*this == other);
151 }
152
153 double getMsec() const
154 {
155 return microseconds_ / 1000.0;
156 }
157
158 double getUsec() const
159 {
160 return static_cast<double>(microseconds_);
161 }
162
163 std::string toString() const {
164 long long absUs = std::abs(microseconds_);
165 long long seconds = absUs / 1'000'000LL;
166 long long fracUs = absUs - seconds * 1'000'000LL;
167 const char *sign = microseconds_ < 0 ? "-" : "";
168
169 using namespace std;
170 ostringstream s;
171 s << sign << seconds << '.' <<
172 setfill('0') << right << setw(6) << fracUs;
173 return s.str();
174 }
175 };

181 class TimeoutMs
182 {
183 public:
184 // arguments: MMTime start time, millisecond interval time
185 explicit TimeoutMs(const MMTime startTime, const unsigned long intervalMs) :
186 startTime_(startTime),
187 interval_(0, 1000*intervalMs)
188 {
189 }
190 explicit TimeoutMs(const MMTime startTime, const MMTime interval) :
191 startTime_(startTime),
192 interval_(interval)
193 {
194 }
195
196 bool expired(const MMTime tnow)
197 {
198 MMTime elapsed = tnow - startTime_;
199 return ( interval_ < elapsed );
200 }
201
202 private:
203 MMTime startTime_; // start time
204 MMTime interval_; // interval in milliseconds
205 };
