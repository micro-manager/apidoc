94inline void DispatchNotification(const Notification& notification,
95 MMEventCallback& cb) {
96 std::visit(detail::overloaded{
97 [&](const notification::PropertiesChanged&) {
98 cb.onPropertiesChanged();
99 },
100 [&](const notification::PropertyChanged& n) {
101 cb.onPropertyChanged(
102 n.deviceLabel.c_str(), n.propertyName.c_str(),
103 n.propertyValue.c_str());
104 },
105 [&](const notification::ConfigGroupChanged& n) {
106 cb.onConfigGroupChanged(
107 n.groupName.c_str(), n.configName.c_str());
108 },
109 [&](const notification::PixelSizeChanged& n) {
110 cb.onPixelSizeChanged(n.pixelSizeUm);
111 },
112 [&](const notification::PixelSizeAffineChanged& n) {
113 cb.onPixelSizeAffineChanged(n.v0, n.v1, n.v2, n.v3, n.v4, n.v5);
114 },
115 [&](const notification::StagePositionChanged& n) {
116 cb.onStagePositionChanged(n.deviceLabel.c_str(), n.position);
117 },
118 [&](const notification::XYStagePositionChanged& n) {
119 cb.onXYStagePositionChanged(
120 n.deviceLabel.c_str(), n.x, n.y);
121 },
122 [&](const notification::ExposureChanged& n) {
123 cb.onExposureChanged(n.deviceLabel.c_str(), n.exposure);
124 },
125 [&](const notification::SLMExposureChanged& n) {
126 cb.onSLMExposureChanged(n.deviceLabel.c_str(), n.exposure);
127 },
128 [&](const notification::ShutterOpenChanged& n) {
129 cb.onShutterOpenChanged(n.deviceLabel.c_str(), n.open);
130 },
131 [&](const notification::ImageSnapped& n) {
132 cb.onImageSnapped(n.cameraLabel.c_str());
133 },
134 [&](const notification::SequenceAcquisitionStarted& n) {
135 cb.onSequenceAcquisitionStarted(n.cameraLabel.c_str());
136 },
137 [&](const notification::SequenceAcquisitionStopped& n) {
138 cb.onSequenceAcquisitionStopped(n.cameraLabel.c_str());
139 },
140 [&](const notification::SystemConfigurationLoaded&) {
141 cb.onSystemConfigurationLoaded();
142 },
143 [&](const notification::ChannelGroupChanged& n) {
144 cb.onChannelGroupChanged(n.channelGroupName.c_str());
145 },
146 }, notification);
147}
