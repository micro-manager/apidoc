72 // tolerate empty configuration names

76 typename std::map<std::string, T>::const_iterator it = configs_.find(oldConfigName);
77 if (it == configs_.end())

90 // tolerate empty configuration names

92 return true;
93
94 typename std::map<std::string, T>::const_iterator it = configs_.find(configName);
95 if (it == configs_.end())
96 return false;
97 configs_.erase(configName);
98 return true;

100
104 bool Delete(const char* configName, const char* deviceLabel, const char* propName)
105 {
106 // tolerate empty configuration names
107 if (strlen(configName) == 0)
108 return true;
109
110 // Check if configuration with configName exists:
111 typename std::map<std::string, T>::const_iterator it = configs_.find(configName);
112 if (it == configs_.end())
113 return false;
114
115 // Delete the specified property
116 configs_[configName].deleteSetting(deviceLabel,propName);
117 return true;
118 }
119
123 std::vector<std::string> GetAvailable() const
124 {
125 std::vector<std::string> configList;
126 typename std::map<std::string, T>::const_iterator it = configs_.begin();
127 while(it != configs_.end())
128 configList.push_back(it++->first);
129
130 return configList;
