19class NotificationQueue {
20public:
21 void Push(Notification notification) {
22 {
23 std::lock_guard<std::mutex> lock(mutex_);
24 queue_.push_back(std::move(notification));
25 }
26 cv_.notify_one();
27 }
28
29 std::optional<Notification> WaitAndPop() {
30 std::unique_lock<std::mutex> lock(mutex_);
31 cv_.wait(lock,
32 [this] { return interrupted_ || !queue_.empty(); });
33 if (interrupted_) {
34 interrupted_ = false;
35 return std::nullopt;
36 }
37 Notification n = std::move(queue_.front());
38 queue_.pop_front();
39 return n;
40 }
41
42 // Wake WaitAndPop (returns nullopt) without discarding pending items.
43 // The flag is consumed by the WaitAndPop call that returns nullopt.
44 void RequestInterrupt() {
45 {
46 std::lock_guard<std::mutex> lock(mutex_);
47 interrupted_ = true;
48 }
49 cv_.notify_one();
50 }
51
52private:
53 std::mutex mutex_;
54 std::condition_variable cv_;
55 std::deque<Notification> queue_;
56 bool interrupted_ = false;
57};
