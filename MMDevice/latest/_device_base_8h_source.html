786 virtual bool GetErrorText(int errorCode, char* text) const
787 {
788 std::map<int, std::string>::const_iterator it;
789 it = messages_.find(errorCode);
790 if (it == messages_.end())
791 {
792 // generic message
793 std::ostringstream osTxt;
794 osTxt << "Error code " << errorCode << " (" << std::setbase(16) << errorCode << " hex)";
795 CDeviceUtils::CopyLimitedString(text, osTxt.str().c_str());
796 return false; // message text not found
797 }
798 else
799 {
800 std::ostringstream stringStreamMessage;
801 stringStreamMessage << it->second.c_str();
802 // add the additional 'property' error info.
803 if( 2<=errorCode && errorCode<=5 )
804 {
805 stringStreamMessage << ": " << GetMorePropertyErrorInfo();
806 }
807 SetMorePropertyErrorInfo("");
808 // native message
809 CDeviceUtils::CopyLimitedString(text, stringStreamMessage.str().c_str());
810 return true; // message found
811 }
812 }

1943 // This converts an absolute position (x_um, y_um), under the current

1948 bool mirrorX, mirrorY;
1949 GetOrientation(mirrorX, mirrorY);
1950
1951 long xSteps{}, ySteps{};
1952 if (mirrorX)
1953 xSteps = originXSteps_ - nint (x_um / this->GetStepSizeXUm());
1954 else
1955 xSteps = originXSteps_ + nint (x_um / this->GetStepSizeXUm());
1956 if (mirrorY)
1957 ySteps = originYSteps_ - nint (y_um / this->GetStepSizeYUm());
1958 else
1959 ySteps = originYSteps_ + nint (y_um / this->GetStepSizeYUm());

1965 // adapter origin and x/y mirroring, to an absolute um position. Do not use
1966 // for relative offsets.

1968 {
1969 bool mirrorX, mirrorY;
1970 GetOrientation(mirrorX, mirrorY);
1971
1972 double x_um{}, y_um{};
1973 if (mirrorX)
1974 x_um = (originXSteps_ - xSteps) * this->GetStepSizeXUm();
1975 else

1977

1980 else

1982
1983 return {x_um, y_um};
