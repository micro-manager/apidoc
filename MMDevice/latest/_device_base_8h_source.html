740 virtual bool GetErrorText(int errorCode, char* text) const
741 {
742 std::map<int, std::string>::const_iterator it;
743 it = messages_.find(errorCode);
744 if (it == messages_.end())
745 {
746 // generic message
747 std::ostringstream osTxt;
748 osTxt << "Error code " << errorCode << " (" << std::setbase(16) << errorCode << " hex)";
749 CDeviceUtils::CopyLimitedString(text, osTxt.str().c_str());
750 return false; // message text not found
751 }
752 else
753 {
754 std::ostringstream stringStreamMessage;
755 stringStreamMessage << it->second.c_str();
756 // add the additional 'property' error info.
757 if( 2<=errorCode && errorCode<=5 )
758 {
759 stringStreamMessage << ": " << GetMorePropertyErrorInfo();
760 }
761 SetMorePropertyErrorInfo("");
762 // native message
763 CDeviceUtils::CopyLimitedString(text, stringStreamMessage.str().c_str());
764 return true; // message found
765 }
766 }
